cmake_minimum_required(VERSION 3.18)
project(cuda_examples LANGUAGES CUDA CXX)

# Enable CUDA
enable_language(CUDA)
find_package(CUDAToolkit REQUIRED)

if (CUDAToolkit_FOUND)
  message(STATUS "CUDA found on this system.")
endif()
message(STATUS "${CUDAToolkit_FOUND}")

# DEBUG: print all the variables that starts with "CUDAToolkit_"
if (ON)
  get_cmake_property(_variableNames VARIABLES)
  foreach (_variableName ${_variableNames})
    if(_variableName MATCHES "^CUDAToolkit_")
      message(STATUS "${_variableName}=${${_variableName}}")
    endif()
  endforeach()
endif()

# Set GoogleTest and CUTLASS include directories from the submodules
set(GOOGLETEST_INCLUDE_DIR ${CMAKE_SOURCE_DIR}/googletest/googletest/include)
set(CUTLASS_INCLUDE_DIR ${CMAKE_SOURCE_DIR}/cutlass/include)

# Common include directories for CUTLASS examples
set(COMMON_INCLUDE_DIRS
    ${CUTLASS_INCLUDE_DIR}
    ${CMAKE_SOURCE_DIR}/cutlass/tools/util/include
    ${CMAKE_SOURCE_DIR}/cutlass/test/unit/gemm/device
    ${GOOGLETEST_INCLUDE_DIR}
)

# Detect GPU architecture dynamically
execute_process(
    COMMAND ${CUDAToolkit_TARGET_DIR}/bin/__nvcc_device_query
    OUTPUT_VARIABLE GPU_ARCH
    OUTPUT_STRIP_TRAILING_WHITESPACE
)

if(GPU_ARCH)
  set(CMAKE_CUDA_ARCHITECTURES ${GPU_ARCH})
else()
  set(CMAKE_CUDA_ARCHITECTURES 80)
endif()

message(STATUS "Detected GPU architecture: ${GPU_ARCH}")

# Common CUDA compiler options
set(CUDA_COMPILE_OPTIONS
    --expt-relaxed-constexpr
    -lineinfo
    --verbose
    --keep
)

# Add binary_instrumentation example (special case)
add_executable(binary_instrumentation binary_instrumentation.cu)
set_target_properties(binary_instrumentation PROPERTIES 
    CUDA_SEPARABLE_COMPILATION ON
    CXX_STANDARD 17
)
target_compile_options(binary_instrumentation PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:${CUDA_COMPILE_OPTIONS}>
)

# Define the list of CUTLASS example targets
set(CUTLASS_EXAMPLES
    halfA_int8_mixed
    sgemm_sm80_cris
    gtc2020_tensorcore_primer
    TMA_1D_arrays_7_29_1
)

# Process all CUTLASS examples in a loop
foreach(target ${CUTLASS_EXAMPLES})
    # Add the executable
    add_executable(${target} ${target}.cu)
    
    # Add include directories
    target_include_directories(${target} PRIVATE ${COMMON_INCLUDE_DIRS})
    
    # Link GoogleTest library
    target_link_libraries(${target} PRIVATE ${CMAKE_BINARY_DIR}/lib/libgtest.a)
    
    # Add CUDA compile options
    target_compile_options(${target} PRIVATE
        $<$<COMPILE_LANGUAGE:CUDA>:${CUDA_COMPILE_OPTIONS}>
    )
endforeach()
